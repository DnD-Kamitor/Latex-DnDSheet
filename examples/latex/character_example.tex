\section*{Features \& Traits}

\subsection*{Racial Traits (Mountain Dwarf)}

\begin{itemize}
    \item \textbf{Darkvision:} 60 feet
    \item \textbf{Dwarven Resilience:} Advantage on saves vs. poison, resistance to poison damage
    \item \textbf{Dwarven Armor Training:} Proficiency with light and medium armor
    \item \textbf{Tool Proficiency:} Smith's tools
    \item \textbf{Stonecunning:} Double proficiency on History checks related to stonework
\end{itemize}

\subsection*{Class Features (Fighter)}

\begin{itemize}
    \item \textbf{Fighting Style:} Defense (+1 AC while wearing armor)
    \item \textbf{Second Wind:} Bonus action to regain 1d10+5 HP (once per short rest)
    \item \textbf{Action Surge:} Take an additional action (once per short rest)
    \item \textbf{Extra Attack:} Attack twice when taking the Attack action
    \item \textbf{Martial Archetype:} Battle Master
\end{itemize}

\subsection*{Battle Master Maneuvers}

\begin{DndSidebar}{Superiority Dice}
    You have \textbf{4 superiority dice} (d8).

    You regain all expended superiority dice when you finish a short or long rest.
\end{DndSidebar}

\textbf{Known Maneuvers:}
\begin{itemize}
    \item \textbf{Riposte:} When a creature misses you with a melee attack, use reaction to make a melee weapon attack, add superiority die to damage
    \item \textbf{Precision Attack:} Add superiority die to attack roll
    \item \textbf{Trip Attack:} Add superiority die to damage, target makes STR save (DC 15) or is knocked prone
\end{itemize}
